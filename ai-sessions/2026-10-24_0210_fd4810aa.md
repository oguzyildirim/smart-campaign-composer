// except that ctx's own error is returned as is when ctx ended.
func (c *Client) Send(ctx context.Context, msg delivery.Message) (string, error) {
	var (
		path string
		body any
	)
	switch msg.Channel {
	case content.ChannelSMS:
		path, body = "/v1/sms", smsRequest{To: msg.To, Body: msg.Body}
	case content.ChannelEmail:
		path, body = "/v1/email", emailRequest{To: msg.To, Subject: msg.Subject, Body: msg.Body}
	default:
		return "", &delivery.ProviderError{Code: delivery.CodeRejected, Err: fmt.Errorf("unknown channel %q", msg.Channel)}
	}
	payload, err := json.Marshal(body)
	if err != nil {
		return "", &delivery.ProviderError{Code: delivery.CodeRejected, Err: err}
	}
	req, err := http.NewRequestWithContext(ctx, http.MethodPost, c.base.JoinPath(path).String(), bytes.NewReader(payload))
	if err != nil {
		return "", &delivery.ProviderError{Code: delivery.CodeRejected, Err: err}
	}
	req.Header.Set("Content-Type", "application/json")
	req.Header.Set("Idempotency-Key", string(msg.IdempotencyKey))

	resp, err := c.http.Do(req)
	if err != nil {
		if ctx.Err() != nil {
			return "", ctx.Err()
		}
		// Timeouts and connection errors: the message may or may not have
		// arrived, so the retry must reuse the idempotency key.
		return "", &delivery.ProviderError{Code: delivery.CodeTimeout, Transient: true, Err: err}
	}
	defer resp.Body.Close()
	data, err := io.ReadAll(io.LimitReader(resp.Body, maxResponseBytes))
	if err != nil {
		if ctx.Err() != nil {
			return "", ctx.Err()
		}
		return "", &delivery.ProviderError{Code: delivery.CodeTimeout, Transient: true, Err: err}
	}

	if resp.StatusCode >= 200 && resp.StatusCode < 300 {
		var r sendResponse
		if err := json.Unmarshal(data, &r); err != nil || r.MessageID == "" || len(r.MessageID) > 128 {
			return "", &delivery.ProviderError{Code: delivery.CodeInvalidResponse, Transient: true,

		}
		return r.MessageID, nil
	}

// Row is one aggregate of a campaign's deliveries: how many have a channel
// and status (and, for failed ones, an error code), and how many of those
// were sent within the window.
type Row struct {
	Channel    content.Channel
	Status     delivery.Status
	ErrorCode  string
	Count      int64
	RecentSent int64
}

// UnknownFailure labels failed deliveries without an error code.
const UnknownFailure = "unknown"

var channelOrder = []content.Channel{content.ChannelSMS, content.ChannelEmail}

// Assemble builds a snapshot from aggregate rows. Channels are the
// campaign's selected channels plus any channel that has deliveries, in
// a fixed order, so a channel never disappears from a dashboard.
func Assemble(id campaign.ID, status campaign.Status, selected []content.Channel, rows []Row, window time.Duration, at time.Time) Snapshot {
	byChannel := map[content.Channel]*ChannelProgress{}
	for _, ch := range selected {
		byChannel[ch] = &ChannelProgress{Channel: ch}
	}
	failures := map[string]int64{}
	for _, r := range rows {
		cp := byChannel[r.Channel]
		if cp == nil {
			cp = &ChannelProgress{Channel: r.Channel}
			byChannel[r.Channel] = cp
		}
		cp.Counts.Add(r.Status, r.Count)
		cp.RecentSent += r.RecentSent
		if r.Status == delivery.StatusFailed {
			code := r.ErrorCode
			if code == "" {
				code = UnknownFailure
			}
			failures[code] += r.Count
		}
	}

	s := Snapshot{CampaignID: id, CampaignStatus: status, Window: window, GeneratedAt: at}
	for _, ch := range channelOrder {
		if cp := byChannel[ch]; cp != nil {
			s.Channels = append(s.Channels, *cp)
		}
	}
	for code, n := range failures {
		s.Failures = append(s.Failures, FailureCount{Code: code, Count: n})
	}
	slices.SortFunc(s.Failures, func(a, b FailureCount) int {
		if a.Count != b.Count {
			return int(b.Count - a.Count)
		}
		return strings.Compare(a.Code, b.Code)
	})
	return s
}

export class FakeEventSource {

  static readonly OPEN = 1

  readonly url: string
  readyState = FakeEventSource.CONNECTING

  onerror: ((ev: Event) => void) | null = null
  private listeners = new Map<string, Set<(ev: MessageEvent) => void>>()

  constructor(url: string | URL) {
    this.url = url.toString()
    FakeEventSource.instances.push(this)
  }

  addEventListener(name: string, fn: (ev: MessageEvent) => void) {
    const set = this.listeners.get(name) ?? new Set()
    set.add(fn)
    this.listeners.set(name, set)
  }

  removeEventListener(name: string, fn: (ev: MessageEvent) => void) {
    this.listeners.get(name)?.delete(fn)
  }

  close() {
    this.readyState = FakeEventSource.CLOSED
  }

  get closed(): boolean {
    return this.readyState === FakeEventSource.CLOSED
  }

  open() {
    this.readyState = FakeEventSource.OPEN
    this.onopen?.(new Event('open'))
  }

  emit(name: string, data: unknown, lastEventId = '') {
    const ev = new MessageEvent(name, { data: typeof data === 'string' ? data : JSON.stringify(data), lastEventId })
    this.listeners.get(name)?.forEach((fn) => fn(ev))
  }

  // error simulates a dropped connection; with retry the browser keeps
  // reconnecting (CONNECTING), without it the source is closed for good.
  error(retry: boolean) {
    this.readyState = retry ? FakeEventSource.CONNECTING : FakeEventSource.CLOSED
    this.onerror?.(new Event('error'))

# Smart Campaign Composer

Marketers describe a campaign in natural language. Claude proposes an
audience segment and SMS/email drafts. A human reviews, edits, and approves,
and approved messages go only to fake users through a local mock provider.
**No real SMS or email is ever sent.**

> **Status: Milestone 5 (queue and reliable delivery).** The stack starts
> with ~5000 seeded fake users. In the browser, a marketer writes a brief
> (Turkish or English), sees the proposed audience with per-channel counts,
> watches SMS and email drafts stream in, edits them, previews them
> personalized for real (fake) recipients, and explicitly approves. Approval
> is persisted, bound to exactly the reviewed content, and creates one
> delivery per consented recipient and channel. Workers send them through
> Kafka to the local mock provider, which fails, delays, and rate-limits on
> purpose, with retries, idempotency keys, and a shared limit of at most 100
> messages per second per channel. Claude or a deterministic mock does the
> AI work. The live dashboard and metrics are not implemented yet.

## Requirements

- Docker with Docker Compose v2 (required for the stack **and** for
  `make test`, which uses testcontainers)
- Go ≥ 1.21 (the pinned go1.27.3 toolchain is downloaded automatically)
- Node.js ≥ 24 and npm (frontend tests and lint)

## Delivery

```
approve ──(one transaction)──► campaigns.status = approved
                               deliveries: one row per consented (user, channel), status queued
                               outbox: one job message per delivery
worker: outbox relay ──► Kafka deliveries.sms / deliveries.email (acks=all)
worker: consumer ──► claim row (lease) ──► rate-limit token ──► consent + render + validate
                 ──► POST mock-provider (Idempotency-Key = delivery id) ──► record outcome
                 ──► commit offset
worker: maintenance (every 1 s) ──► expired leases → retrying; due retries → queued + outbox;
                                    campaigns with only sent/failed deliveries → completed
```

- **One delivery per person, channel, and campaign.** `deliveries` has
  `UNIQUE (campaign_id, user_id, channel)`. Approving again creates nothing.

## Milestone 6 — Live monitoring

### D48. Periodic aggregate snapshots, not per-row events

The dashboard shows a snapshot: per-channel counts for all five delivery
statuses, failures by error code, recent sends, and the campaign status,
read in one statement (`campaigns LEFT JOIN LATERAL (… GROUP BY channel,
status, failed code)`), so status and counts are mutually consistent. It
uses the existing `deliveries (campaign_id, status)` index from M5, so no
migration was needed.

Alternatives considered:

- **Streaming every delivery change** (per-row events from the worker or
  Kafka): cost grows with send volume (up to 200 events/s per campaign),
  clients must apply increments in order, and a missed or repeated event
  silently corrupts the numbers. Reconnection needs replay.
- **Postgres `LISTEN/NOTIFY` → SSE**: lower latency, but a trigger or a
  notify on every delivery update adds write-path cost to the hot loop,
  payloads are capped at 8 KB, and notifications are lost while the API is
  disconnected, so a periodic re-read is still needed for correctness.
- **Snapshots on a timer (chosen)**: bounded cost (one indexed query per
  campaign per interval), full state in every event, so reconnects and
  duplicates are harmless. Cost: up to one interval (default 1 s) of lag.

### D49. One shared poller per campaign

`monitoring.Hub` starts one poll loop for a campaign when its first viewer
subscribes and stops it when the last one leaves, so N viewers cost one
query per interval, not N. Each subscription holds only the latest
snapshot (a slow client skips intermediate ones; every snapshot is full
state). The SSE handler sends a snapshot only when it differs from the
last one it sent (ignoring `generated_at`). A per-campaign viewer limit
(`PROGRESS_MAX_VIEWERS`, 429 beyond it) bounds memory and goroutines.

A new viewer reads one direct snapshot on connect, which also turns an
unknown campaign into a 404 before the stream starts. Subscribing happens
first, so no change between the two is missed. Pollers are per API
process; with several API instances each polls once per campaign it has
viewers for.

`http.Server.Shutdown` does not cancel request contexts, so open streams

streams end and the browser reconnects later.

**What was built**
- **Progress snapshot:** one Postgres query returns per-channel counts for all five delivery statuses, failures by reason, recent sends, and the campaign status. It uses the `(campaign_id, status)` index that already existed, so no migration.
- **Shared poller:** all viewers of a campaign share one ~1s poll, so two tabs cost one query, not two. It stops when the last viewer leaves, and there's a viewer limit per campaign (429 beyond it).
- **Endpoints:** `GET /api/v1/campaigns/{id}/progress` returns the snapshot as JSON. `GET …/events` streams it: a full snapshot on connect, then only when something changed, then a final `complete` event and the stream closes. Reconnecting gets a fresh snapshot and the event ids keep increasing. On API shutdown the streams close right away instead of holding shutdown up.
- **Metrics:** `/metrics` on the api and each worker, with the metrics the prompt listed. Labels only take fixed values (route patterns, channels, error codes), never campaign, user or delivery ids. Prometheus code stays out of the domain packages.
- **Health:** the worker and mock provider share one probe router, the mock provider now has `/readyz`, and Compose healthchecks use `/readyz`. Each worker's port 8081 is published on a random host port, so `--scale worker=3` still works.
- **Frontend:** a dashboard page with per-channel counts, accessible progress bars, throughput, failures by reason, and a live / reconnecting / disconnected / completed indicator. Approval redirects there, and the review page and campaign list link to it.
- **Docs:** README (dashboard, metrics, example PromQL), DECISIONS D48–D52, the CLAUDE.md layout section, `.env.example` (three `PROGRESS_*` settings), and nginx (`/api/metrics` blocked from the browser).

**Code review (Standards and Spec, run separately)**

Fixed:
- The SQL used `clock_timestamp()`, so the recent-sends filter and `generated_at` could disagree within one read. It now uses `now()`.
- The frontend recomputed per-channel percent, duplicating a backend rule. The backend now returns `percent_done` per channel.
- `http_request_duration_seconds` lacked `status_class` (the spec asks for it) and recorded whole SSE sessions as latency. Both fixed.
- Minor: the failure sort now uses `cmp.Compare`, and rate-limit wait is recorded only when a token was actually obtained.

Left open:
- Small duplication with the existing handlers (campaign id parsing, error mapping, `event` vs `eventWithID`).
- LLM outcome strings are untyped: the `invalid_output` literal is copied into the metrics package, so renaming it in the generator would silently break `llm_validation_failures_total`.
- `useCampaignProgress` opens the EventSource itself instead of going through the API layer.
- The events endpoint starts a poller briefly before it knows the campaign exists (cleaned up when the 404 returns).
- If a campaign is deleted while someone is watching, its poller keeps logging warnings until they leave.
- `outbox_pending` is reported by every worker, so aggregate it with `max`, not `sum` (in the README).
- Scope additions beyond the prompt: a `released` delivery outcome, separate `429`/`error` provider classes, `too_many_viewers`, and the `PROGRESS_*` settings.
